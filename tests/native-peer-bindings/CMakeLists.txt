cmake_minimum_required(VERSION 3.24)
project(PeerBindingsHarness LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.7 REQUIRED COMPONENTS Core)

target_include_directories(peer-bindings-harness PRIVATE "${DESKFLOW_SOURCE}/src/lib")
target_link_libraries(peer-bindings-harness PRIVATE Qt6::Core)
