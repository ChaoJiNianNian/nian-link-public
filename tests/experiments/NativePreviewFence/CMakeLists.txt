cmake_minimum_required(VERSION 3.24)
project(NativePreviewFenceTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DESKFLOW_SOURCE OR NOT DESKFLOW_BUILD)
  message(FATAL_ERROR "Set DESKFLOW_SOURCE and DESKFLOW_BUILD to an isolated, verified source and build.")
endif()
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Network)
add_executable(native-fence-unit fence_tests.cpp PreviewFenceState.cpp)
add_executable(native-fence-queue queue_harness.cpp PreviewFenceState.cpp)
add_executable(native-fence-async-queue async_queue_harness.cpp PreviewFenceState.cpp)
foreach(target IN ITEMS native-fence-queue native-fence-async-queue)
  target_include_directories(${target} PRIVATE "${DESKFLOW_SOURCE}/src/lib" "${DESKFLOW_BUILD}/src/lib")
  foreach(lib IN ITEMS base mt arch common)
    target_link_libraries(${target} PRIVATE "${DESKFLOW_BUILD}/src/lib/${lib}/${lib}.lib")
  endforeach()
  target_link_libraries(${target} PRIVATE Qt6::Core Qt6::Network ws2_32 userenv Wtsapi32 wininet shlwapi version crypt32)
endforeach()
